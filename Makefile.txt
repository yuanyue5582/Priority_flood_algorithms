# Makefile for DEM_Filling project
# Compiler and flags
CXX       = g++
CXXFLAGS  = -std=c++11 -Wall -O2 -MMD -MP
LDFLAGS   =
LDLIBS    = -lgdal

# Use pkg-config to get GDAL flags if available
ifeq ($(shell pkg-config --exists gdal && echo yes), yes)
    CXXFLAGS += $(shell pkg-config --cflags gdal)
    LDLIBS   += $(shell pkg-config --libs gdal)
else
    # Fallback: assume GDAL headers in /usr/include/gdal
    CXXFLAGS += -I/usr/include/gdal
endif

# Target executable name
TARGET = DEM_Filling

# Source files (all .cpp files in the project)
SRCS = main.cpp \
       dem.cpp \
       utils.cpp \
       FillDEM_Barnes.cpp \
       FillDEM_PD.cpp \
       FillDEM_Wang.cpp \
       fillDEM_Wei.cpp \
       FillDEM_Zhou_OnePass.cpp \
       FillDEM_Zhou_TwoPass.cpp \
       FillDEM_Zhou_Direct.cpp

# Object files and dependency files
OBJS = $(SRCS:.cpp=.o)
DEPS = $(OBJS:.o=.d)

# Default target
all: $(TARGET)

# Link the executable
$(TARGET): $(OBJS)
	$(CXX) $(LDFLAGS) -o $@ $^ $(LDLIBS)

# Compile C++ source files into object files
%.o: %.cpp
	$(CXX) $(CXXFLAGS) -c $< -o $@

# Include automatically generated dependency files
-include $(DEPS)

# Clean up build artifacts
clean:
	rm -f $(OBJS) $(DEPS) $(TARGET)

# Declare phony targets
.PHONY: all clean
